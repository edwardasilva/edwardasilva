% Unicode mapping, PDF metadata, and bookmarks. scripts/build-cv.js copies it next to two generated
% files before each compile: settings.tex holds the font size, margins, and spacing for one layout,
% and body.tex holds the metadata and content. It holds no CV text.
%
% File Structure:
% - Packages and page setup
% - Section and list style, with the spacing read from settings.tex
% - Header-line macro that sets a date or location flush with the right margin
% - Measurement macro the fitting script reads from the log
%
% Where this file is used within the repository:
% - Compiled by scripts/build-cv.js
%
% Usage:
% $ `npm run build:cv` : Fits and compiles the CV

\documentclass[11pt]{article}
\input{settings}
\usepackage[letterpaper,margin=\resumemargin]{geometry}
\usepackage{XCharter}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
% The less-than-or-equal sign and sigma in a bullet, typeset from the math font.
\DeclareUnicodeCharacter{2264}{\ensuremath{\leq}}
\DeclareUnicodeCharacter{03C3}{\ensuremath{\sigma}}
\usepackage{enumitem}
\usepackage{array}
\usepackage[hidelinks]{hyperref}
\usepackage{bookmark}
\usepackage{titlesec}
\usepackage[protrusion=true,expansion=true]{microtype}
\raggedright
\raggedbottom
\pagestyle{empty}
\urlstyle{same}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% No hyphenated line breaks: a word split across lines cannot be found by a keyword search.
\hyphenpenalty=10000
\exhyphenpenalty=10000

% No paragraph or bullet leaves a single line alone at the bottom or the top of a page.
\clubpenalty=10000
\widowpenalty=10000

% Every glyph maps to real Unicode text, so the PDF stays machine-readable.
\input{glyphtounicode}
\pdfgentounicode=1

% One gap size between every pair of sections, grown with the rest of the spacing.
\titleformat{\section}{\bfseries\large}{}{0pt}{}[\vspace{2pt}\titlerule]
\titlespacing*{\section}{0pt}{\sectiongap}{\sectionafter}

\renewcommand\labelitemi{$\vcenter{\hbox{\small$\bullet$}}$}
% An entry never splits across a page: no page break before its first bullet or between bullets.
\setlist[itemize]{leftmargin=12pt,topsep=\listtop,itemsep=\listitem,parsep=0pt,partopsep=0pt,beginpenalty=10000,midpenalty=10000}

% One header line: the left text, then the right text (a date or a location) flush with the right
% edge of the text block. \raggedright gives the last line of a paragraph a stretchable
% \parfillskip that can pull \hfill text off the margin, so the line sets \parfillskip to zero.
% Character protrusion is off for that line, so a final letter such as the t in Present ends at
% the margin, where the section rule ends, instead of hanging a fraction of a point past it.
% When the left text and the right text do not fit on one line with a 2 em gap, the left text
% wraps in its own column of a top-aligned tabular, and the right text stays on the first line. With no right text the left text prints as
% a plain paragraph.
\newlength{\cvleftwidth}
\newlength{\cvrightwidth}
\newcommand{\cvline}[2]{%
  \par\noindent
  \if\relax\detokenize{#2}\relax
    #1%
  \else
    \settowidth{\cvleftwidth}{#1}%
    \settowidth{\cvrightwidth}{#2}%
    \ifdim\dimexpr\cvleftwidth+\cvrightwidth+2em\relax>\linewidth
      \begin{tabular}[t]{@{}>{\raggedright\arraybackslash}p{\dimexpr\linewidth-\cvrightwidth-2em\relax}@{\hspace{2em}}r@{}}%
        #1 & #2%
      \end{tabular}%
    \else
      #1\hfill#2%
    \fi
  \fi
  {\parfillskip=0pt\pdfprotrudechars=0\par}}

% The fitting script reads this line from the log after each compile.
\newcommand{\measure}[1]{%
  \typeout{CV-#1 page=\thepage\space used=\the\pagetotal\space
    goal=\the\pagegoal\space line=\the\baselineskip}}

\begin{document}
\resumefont
\thispagestyle{empty}
\input{body}
\par\measure{END}
\end{document}
